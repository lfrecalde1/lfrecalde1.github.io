\section{Research Focus}
{\small I am a research engineer with a Master of Science in Control Systems Engineering and a bachelor's degree in Mechatronics Engineering.
My research aims to enable \textbf{fully onboard autonomous navigation in unstructured environments}.
I pursue this goal by developing control and estimation algorithms using \textbf{nonlinear control theory, differential geometry, and numerical optimization}.
My work spans dynamical modeling, physics-based simulation, and \textbf{real-world experiments}.\par}

\section{Technical Skills}
{\small\textbf{Optimization}\par}
\begin{itemize}[leftmargin=0.22in, itemsep=2pt, parsep=0pt, topsep=4pt]
\small
\item \textbf{Model predictive control (MPC):} \href{https://arxiv.org/abs/2510.11525}{Nonlinear model predictive control (NMPC)} and \href{https://arxiv.org/abs/2609.36482}{model predictive contouring control (MPCC)}; sequential quadratic programming (SQP) and its real-time iteration scheme (SQP-RTI), interior-point methods, differential dynamic programming (DDP), and iterative linear quadratic regulator (iLQR).
\item \textbf{Nonsmooth and hybrid optimal control:} complementarity-constrained formulations and relaxation methods for trajectory optimization across mode transitions.
\item \textbf{Manifold optimization:} Geometric formulations for \href{https://arxiv.org/abs/2410.13541}{robot pose estimation} and \href{https://arxiv.org/abs/2510.11525}{control}.
\item \textbf{Differentiable optimization:} Implicit differentiation of Karush--Kuhn--Tucker (KKT) conditions to compute solution sensitivities for gradient-based MPC tuning and learning.
\item \textbf{Distributed estimation and control:} Alternating direction method of multipliers (ADMM) for distributed sensor fusion and multi-robot coordination.
\end{itemize}
{\small\textbf{Modeling and Control}\par}
\begin{itemize}[leftmargin=0.22in, itemsep=2pt, parsep=0pt, topsep=4pt]
\small
\item \textbf{Geometric control on Lie groups:} Controller design and \href{https://arxiv.org/abs/2504.08841}{stability analysis} for \href{https://arxiv.org/abs/2510.11525}{attitude and pose tracking}.
\item \textbf{Task-priority control:} Null-space projection and affine-subspace methods for prioritizing robotic tasks (\href{https://arxiv.org/abs/2311.08019}{unmanned aerial vehicle (UAV) inspection}; \href{https://arxiv.org/abs/2605.20392}{tactile manipulation}).
\item \textbf{Safety-critical control:} Control Lyapunov functions (\href{https://arxiv.org/abs/2504.08841}{CLFs}) and control barrier functions (\href{https://arxiv.org/abs/2504.08841}{CBFs}) for robotic systems.
\item \textbf{Data-driven modeling and control:} \href{https://doi.org/10.1016/j.ifacol.2023.12.088}{Dynamic mode decomposition with control (DMDc)} and Koopman operator approximations for robot dynamics identification and predictive modeling.
\end{itemize}
